
#+TITLE: Tree-sitter Configuration

* Core Availability Check
We wrap everything in a check to ensure this specific Emacs binary was compiled with Tree-sitter support.

#+begin_src emacs-lisp
;; -*- mode: emacs-lisp; lexical-binding: t -*-
(when (and (fboundp 'treesit-available-p)
           (treesit-available-p))
  (add-to-list 'treesit-extra-load-path
               (expand-file-name "tree-sitter" user-emacs-directory))
  (use-package treesit-auto
    :demand t
    :custom
    (treesit-auto-install 'prompt)
    (treesit-auto-langs '(go rust python toml haskell))
    :config
    (global-treesit-auto-mode))
  (add-hook 'python-ts-mode-hook
            (lambda ()
              (run-hooks 'python-mode-hook 'python-mode-local-vars-hook
                         'python-mode-base-hook)))
  (add-hook 'typescript-tsx-mode-hook
            (lambda ()
              (run-hooks 'typescript-mode-hook)))
  (add-hook 'python-ts-mode-hook #'lsp-deferred)
  ;; cuda
  (setq treesit-language-source-alist
        (append treesit-language-source-alist
                '((toml "https://github.com/tree-sitter/tree-sitter-toml")
                  (cuda "https://github.com/tree-sitter-grammars/tree-sitter-cuda")
                  (tree-sitter "https://github.com/tree-sitter/tree-sitter")
                  (haskell "https://github.com/lesliebinbin/tree-sitter-haskell"))))
  ;; cuda
  (add-to-list 'auto-mode-alist '("\\.cu\\'" . cuda-ts-mode))
  (add-to-list 'auto-mode-alist '("\\.cuh\\'" . cuda-ts-mode))
  (define-derived-mode cuda-ts-mode c++-mode "CUDA[TS]"
    (when (treesit-ready-p 'cuda)
      (treesit-parser-create 'cuda)
      (treesit-major-mode-setup)))
  (add-hook 'cuda-ts-mode-hook #'spacemacs//c-c++-setup-backend)
  (with-eval-after-load 'lsp-mode (add-to-list 'lsp-language-id-configuration '(cuda-ts-mode . "cuda")))
  (spacemacs|define-jump-handlers cuda-ts-mode)
  ;;cuda
  ;;toml
  (when (treesit-ready-p 'toml)
    (add-to-list 'major-mode-remap-alist '(toml-mode . toml-ts-mode)))
  ;;toml
  ;;python
  (when (treesit-ready-p 'python)
    (add-to-list 'major-mode-remap-alist '(python-mode . python-ts-mode)))
  ;;python
  )
